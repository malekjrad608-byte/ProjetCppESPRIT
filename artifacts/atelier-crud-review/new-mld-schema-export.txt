  CREATE TABLE "SERVICE"
   (	"ID_SERV" NUMBER(10,0),
	"REF_SERVICE" VARCHAR2(120 CHAR) NOT NULL ENABLE,
	"PRESTATION" CLOB,
	"RESPONSIBLE" VARCHAR2(250 CHAR),
	"CATEGORIE" VARCHAR2(120 CHAR) NOT NULL ENABLE,
	"DATE_HEUR" VARCHAR2(35 CHAR) NOT NULL ENABLE,
	"STATUT" VARCHAR2(30 CHAR) NOT NULL ENABLE,
	"DUREE" NUMBER(4,0) NOT NULL ENABLE,
	"TARIF" NUMBER(10,2) NOT NULL ENABLE,
	 CHECK ("STATUT"='Planifié' OR "STATUT"='En cours' OR "STATUT"='Terminé' OR "STATUT"='Annulé') ENABLE,
	 CHECK ("DUREE">=5 AND "DUREE"<=1440) ENABLE,
	 CHECK ("TARIF">=0 AND "TARIF"<=99999.99) ENABLE,
	 PRIMARY KEY ("ID_SERV")
  USING INDEX  ENABLE,
	 CONSTRAINT "UQ_SERVICE_REF" UNIQUE ("REF_SERVICE")
  USING INDEX  ENABLE
   ) ;


  CREATE TABLE "LOCAL"
   (	"ID_LOCAL" NUMBER(10,0),
	"NOM_LOCAL" VARCHAR2(150 CHAR) NOT NULL ENABLE,
	"TYPE" VARCHAR2(100 CHAR) NOT NULL ENABLE,
	"ETAGE" NUMBER(3,0) NOT NULL ENABLE,
	"CAPACITE" NUMBER(6,0) NOT NULL ENABLE,
	"OCCUPANCY" NUMBER(6,0) NOT NULL ENABLE,
	"SPECIALITE" VARCHAR2(250 CHAR),
	"OUT_OF_SERVICE" NUMBER(1,0) NOT NULL ENABLE,
	"CLEANING" NUMBER(1,0) NOT NULL ENABLE,
	"LAST_CLEANED" VARCHAR2(35 CHAR),
	"DIRTY_SINCE" VARCHAR2(35 CHAR),
	"ID_SERV" NUMBER(10,0) NOT NULL ENABLE,
	"ETAT" VARCHAR2(12) GENERATED ALWAYS AS (CASE  WHEN "OUT_OF_SERVICE"=1 THEN 'Hors service' WHEN "OCCUPANCY">0 THEN 'Occupe' ELSE 'Disponible' END) VIRTUAL ,
	 CHECK ("CAPACITE">=1) ENABLE,
	 CHECK (out_of_service IN (0,1)) ENABLE,
	 CHECK (cleaning IN (0,1,2)) ENABLE,
	 CHECK ("OCCUPANCY">=0 AND "OCCUPANCY"<="CAPACITE") ENABLE,
	 PRIMARY KEY ("ID_LOCAL")
  USING INDEX  ENABLE,
	 CONSTRAINT "UQ_LOCAL_NOM" UNIQUE ("NOM_LOCAL")
  USING INDEX  ENABLE
   ) ;


  CREATE TABLE "EMPLOYER"
   (	"ID_EMP" NUMBER(10,0),
	"NOM" VARCHAR2(120 CHAR) NOT NULL ENABLE,
	"PRENOM" VARCHAR2(120 CHAR) NOT NULL ENABLE,
	"COURRIEL" VARCHAR2(250 CHAR),
	"TEL" VARCHAR2(60 CHAR),
	"POSTE" VARCHAR2(150 CHAR),
	"DATE_EMBAUCHE" VARCHAR2(10 CHAR),
	"DISPONIBLE" NUMBER(1,0) DEFAULT 1,
	"CHARGE" NUMBER(8,2) DEFAULT 0,
	"DISPO_REPOS" NUMBER(8,2) DEFAULT 35,
	"COMPETENCES" VARCHAR2(2000 CHAR),
	"ID_SERV" NUMBER(10,0) NOT NULL ENABLE,
	"CIN" VARCHAR2(30 CHAR),
	"DATE_NAISS" VARCHAR2(10 CHAR),
	"ADRESSE" VARCHAR2(500 CHAR),
	"TYPE_CONTRAT" VARCHAR2(80 CHAR),
	"SALAIRE" NUMBER(12,2),
	 CHECK (disponible IN (0,1)) ENABLE,
	 CHECK ("CHARGE">=0) ENABLE,
	 CHECK ("DISPO_REPOS">=0) ENABLE,
	 PRIMARY KEY ("ID_EMP")
  USING INDEX  ENABLE,
	 CONSTRAINT "UQ_EMPLOYER_CIN" UNIQUE ("CIN")
  USING INDEX  ENABLE,
	 CONSTRAINT "CK_EMPLOYER_SALAIRE" CHECK (salaire>=0) ENABLE
   ) ;


  CREATE TABLE "STOCK"

	"ARTICLE" VARCHAR2(250 CHAR) NOT NULL ENABLE,

	"SEUIL_MIN" NUMBER(6,0) NOT NULL ENABLE,

  CREATE TABLE "MEDICAL"

	"TYPE_ANIMAL" VARCHAR2(30 CHAR) NOT NULL ENABLE,
	"RACE" VARCHAR2(120 CHAR),

	"SEXE" VARCHAR2(20 CHAR) NOT NULL ENABLE,
	"PROPRIETAIRE" VARCHAR2(120 CHAR) NOT NULL ENABLE,

	"DIAGNOSTIC" CLOB,
	"TREATMENT" CLOB,
	"VACCINATIONS" CLOB,
	"ALLERGIES" CLOB,
	"WEIGHT" NUMBER(6,2) NOT NULL ENABLE,
	"OBSERVATIONS" CLOB,
	"VACCINATION_STATUS" VARCHAR2(30 CHAR) NOT NULL ENABLE,

  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "SERVICES" ("ID", "NAME", "DESCRIPTION", "RESPONSIBLE", "CATEGORY", "DATETIME", "STATUS", "DURATION", "PRICE", "CLIENTNAME", "INVOICENUMBER", "INVOICEDATE", "INVOICE_STATUS", "INVOICE_EMITTED_AT") AS
  SELECT id_serv id,ref_service name,prestation description,responsible,
 categorie category,date_heur datetime,statut status,duree duration,tarif price,
 ref_service clientname,CAST(NULL AS VARCHAR2(60)) invoicenumber,
 CAST(NULL AS VARCHAR2(35)) invoicedate,'BROUILLON' invoice_status,
 CAST(NULL AS VARCHAR2(35)) invoice_emitted_at FROM service;


  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "LOCAUX" ("ID", "NAME", "TYPE", "FLOOR", "CAPACITY", "OCCUPANCY", "SPECIALTY", "OUT_OF_SERVICE", "CLEANING", "LAST_CLEANED", "DIRTY_SINCE", "ID_SERVICE", "ETAT") AS
  SELECT id_local id,nom_local name,type,etage floor,capacite capacity,occupancy,
 specialite specialty,out_of_service,cleaning,last_cleaned,dirty_since,id_serv id_service,etat FROM local;


  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "STOCK_LOTS" ("LOT", "ARTICLE", "CATEGORY", "QUANTITY", "UNIT", "THRESHOLD", "EXPIRY", "ID_LOCAL", "ID_ARTICLE", "NOM_ARTICLE") AS
  SELECT lot,article,type_article category,quantite_disp quantity,unit,seuil_min threshold,
 date_echeance expiry,id_local,id_article,nom_article FROM stock;


  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "EMPLOYES" ("ID", "NOM", "PRENOM", "COURRIEL", "TELEPHONE", "POSTE", "DATE_EMBAUCHE", "DISPONIBLE", "CHARGE_HEBDO", "DISPO_HEBDO", "COMPETENCES", "ID_SERVICE", "CIN", "DATE_NAISS", "ADRESSE", "TYPE_CONTRAT", "SALAIRE") AS
  SELECT id_emp id,nom,prenom,courriel,tel telephone,poste,date_embauche,disponible,
 charge charge_hebdo,dispo_repos dispo_hebdo,competences,id_serv id_service,
 cin,date_naiss,adresse,type_contrat,salaire FROM employer;


  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "MEDICAL_RECORDS" ("ID", "ANIMAL_ID", "NAME", "SPECIES", "BREED", "AGE", "SEX", "PROPRIETAIRE", "CONSULTATION_DATE", "REASON", "DIAGNOSIS", "TREATMENT", "VACCINATIONS", "ALLERGIES", "WEIGHT", "OBSERVATIONS", "VACCINATION_STATUS", "ID_EMPLOYE", "ESPECE") AS
  SELECT id_doss id,id_animal animal_id,nom_animal name,type_animal species,race breed,
 age,sexe sex,proprietaire,date_consultation consultation_date,motif reason,
 diagnostic diagnosis,treatment,vaccinations,allergies,weight,observations,
 vaccination_status,id_emp id_employe,espece FROM medical;


  CREATE OR REPLACE FORCE NONEDITIONABLE VIEW "PLANNING_AFFECTATIONS" ("ID", "TITRE", "COMPETENCE_REQUISE", "DEBUT", "FIN", "CHARGE_HEURES", "RESPONSABLE_ID", "CREATED_AT") AS
  SELECT e.id_emp id,s.ref_service titre,NVL(e.competences,'Service') competence_requise,
 REPLACE(SUBSTR(s.date_heur,1,19),'T',' ') debut,
 TO_CHAR(TO_TIMESTAMP(SUBSTR(s.date_heur,1,19),'YYYY-MM-DD"T"HH24:MI:SS')+NUMTODSINTERVAL(s.duree,'MINUTE'),'YYYY-MM-DD HH24:MI:SS') fin,
 s.duree/60 charge_heures,e.id_emp responsable_id,s.date_heur created_at
FROM employer e JOIN service s ON s.id_serv=e.id_serv WHERE s.statut<>'Annulé' WITH READ ONLY;

  CREATE OR REPLACE NONEDITIONABLE TRIGGER "EMPLOYEE_TRANSFER_GUARD"
BEFORE UPDATE OF id_serv ON employer FOR EACH ROW
DECLARE a TIMESTAMP;b TIMESTAMP;da NUMBER;db NUMBER;sa VARCHAR2(30);sb VARCHAR2(30);
BEGIN
 IF :NEW.id_serv<>:OLD.id_serv THEN
  SELECT TO_TIMESTAMP(SUBSTR(date_heur,1,19),'YYYY-MM-DD"T"HH24:MI:SS'),duree,statut INTO a,da,sa FROM service WHERE id_serv=:OLD.id_serv;
  SELECT TO_TIMESTAMP(SUBSTR(date_heur,1,19),'YYYY-MM-DD"T"HH24:MI:SS'),duree,statut INTO b,db,sb FROM service WHERE id_serv=:NEW.id_serv;
  IF sa<>'Annulé' AND sb<>'Annulé' AND a<b+NUMTODSINTERVAL(db,'MINUTE') AND b<a+NUMTODSINTERVAL(da,'MINUTE') THEN
   RAISE_APPLICATION_ERROR(-20051,'Employe deja occupe sur ce creneau');
  END IF;
 END IF;
EXCEPTION WHEN NO_DATA_FOUND THEN
 RAISE_APPLICATION_ERROR(-20052,'Service de destination introuvable');
END;
/
ALTER TRIGGER "EMPLOYEE_TRANSFER_GUARD" ENABLE;
